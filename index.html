""" update_trimika.py ───────────────────────────────────────────────────────────────────────────── Daily pipeline for the Trimika HTS Search Tool: 1. Pull today's XLSX from the wy_trimika_tool email 2. Download trimika_data.xlsx from GitHub 3. Append new rows, deduplicate on REFERENCE NO + 7501 LINE # 4. Push combined file back to GitHub 5. Pull today's EDI CSV from the HTS_SEARCH_TOOL_EDI email 6. Download edi_current.csv from GitHub 7. Append new rows, deduplicate on FILENUMBER + LINENUMBER 8. Push combined file back to GitHub Schedule via comprehensive_scheduler.py at 07:32 every weekday. Author : IFF Automation ───────────────────────────────────────────────────────────────────────────── """ import os import io import base64 import tempfile import logging import requests import pandas as pd import pythoncom import win32com.client # ───────────────────────────────────────────────────────── # CONFIGURATION # ───────────────────────────────────────────────────────── SHARED_MAILBOX = "reports@iffusa.com" REPORTS_FOLDER = "REPORTS" EMAIL_SUBJECT = "wy_trimika_tool" EDI_SUBJECT = "HTS_SEARCH_TOOL_EDI" GITHUB_OWNER = "iffinc-ga" GITHUB_REPO = "hts_search_tool" GITHUB_BRANCH = "main" GITHUB_TOKEN = "" # ← paste your PAT here XLSX_FILE = "trimika_data.xlsx" EDI_FILE = "edi_current.csv" # Dedup keys XLSX_DEDUP_COLS = ["REFERENCE NO", "7501 LINE #"] EDI_DEDUP_COLS = ["BROKERREFERENCE", "LINENUMBER"] ERROR_NOTIFY = "winnie.yim@iffusa.com" # ───────────────────────────────────────────────────────── logger = logging.getLogger(__name__) # ══════════════════════════════════════════════════════════ # GITHUB HELPERS # ══════════════════════════════════════════════════════════ def _gh_headers(): return { "Authorization": f"token {GITHUB_TOKEN}", "Accept": "application/vnd.github.v3+json", } def _gh_api_url(file_path: str) -> str: return f"https://api.github.com/repos/{GITHUB_OWNER}/{GITHUB_REPO}/contents/{file_path}" def download_from_github(file_path: str) -> bytes | None: """ Download a file's raw bytes from GitHub. Returns bytes on success, None if file doesn't exist yet. """ resp = requests.get( f"{_gh_api_url(file_path)}?ref={GITHUB_BRANCH}", headers=_gh_headers(), timeout=30, ) if resp.status_code == 404: logger.info("[Trimika] '%s' not found on GitHub — will create fresh.", file_path) return None if resp.status_code != 200: raise RuntimeError(f"GitHub GET {file_path} → {resp.status_code}: {resp.text[:200]}") data = resp.json() return base64.b64decode(data["content"].replace("\n", "")) def push_to_github(file_path: str, file_bytes: bytes, message: str) -> bool: """ Create or update a file on GitHub. Automatically fetches the current SHA required for updates. """ if not GITHUB_TOKEN: logger.error("[Trimika] GITHUB_TOKEN is not set — cannot push to GitHub") return False api_url = _gh_api_url(file_path) # Get existing SHA if file already exists existing_sha = None resp = requests.get( f"{api_url}?ref={GITHUB_BRANCH}", headers=_gh_headers(), timeout=15, ) if resp.status_code == 200: existing_sha = resp.json().get("sha") logger.info("[Trimika] '%s' current SHA: %s", file_path, existing_sha) elif resp.status_code != 404: logger.error("[Trimika] GitHub GET error %s: %s", resp.status_code, resp.text[:200]) return False payload = { "message": message, "content": base64.b64encode(file_bytes).decode("utf-8"), "branch": GITHUB_BRANCH, } if existing_sha: payload["sha"] = existing_sha put_resp = requests.put(api_url, headers=_gh_headers(), json=payload, timeout=60) if put_resp.status_code in (200, 201): logger.info("[Trimika] ✓ Pushed '%s' to GitHub", file_path) return True else: logger.error("[Trimika] GitHub PUT failed %s: %s", put_resp.status_code, put_resp.text[:400]) return False # ══════════════════════════════════════════════════════════ # OUTLOOK HELPERS # ══════════════════════════════════════════════════════════ def _find_folder(parent, name: str): """Recursively find a mail folder by name (case-insensitive).""" for sub in parent.Folders: if sub.Name.upper() == name.upper(): return sub found = _find_folder(sub, name) if found: return found return None def get_attachment_from_outlook(subject: str, ext: str) -> str | None: """ Find the most recent email in REPORTS_FOLDER whose subject contains `subject`, save the first attachment with extension `ext` to a temp file, mark the email as read, and return the local path. """ pythoncom.CoInitialize() local_path = None try: outlook = win32com.client.Dispatch("Outlook.Application") namespace = outlook.GetNamespace("MAPI") shared_box = None for folder in namespace.Folders: if SHARED_MAILBOX.lower() in folder.Name.lower(): shared_box = folder break if shared_box is None: logger.error("[Trimika] Cannot find shared mailbox '%s'", SHARED_MAILBOX) return None reports_folder = _find_folder(shared_box, REPORTS_FOLDER) if reports_folder is None: top = [f.Name for f in shared_box.Folders] logger.error("[Trimika] Cannot find '%s'. Top-level folders: %s", REPORTS_FOLDER, top) return None items = reports_folder.Items items.Sort("[ReceivedTime]", True) target_msg = None for msg in items: try: if subject.lower() in str(msg.Subject).lower(): target_msg = msg break except Exception: continue if target_msg is None: logger.error("[Trimika] No email found with subject '%s'", subject) return None logger.info("[Trimika] Found: '%s' received %s", target_msg.Subject, target_msg.ReceivedTime) for att in target_msg.Attachments: if str(att.FileName).lower().endswith(ext): tmp = tempfile.NamedTemporaryFile( suffix=ext, delete=False, prefix="trimika_" ) tmp.close() att.SaveAsFile(tmp.name) local_path = tmp.name logger.info("[Trimika] Saved '%s' → %s", att.FileName, local_path) try: target_msg.UnRead = False target_msg.Save() logger.info("[Trimika] Marked email as read") except Exception as e: logger.warning("[Trimika] Could not mark as read: %s", e) break if local_path is None: logger.error("[Trimika] Email found but no %s attachment", ext) except Exception as exc: logger.exception("[Trimika] Outlook error: %s", exc) finally: pythoncom.CoUninitialize() return local_path # ══════════════════════════════════════════════════════════ # XLSX: APPEND + DEDUP + PUSH # ══════════════════════════════════════════════════════════ def append_and_push_xlsx(new_file_path: str) -> bool: """ Download trimika_data.xlsx from GitHub, append today's rows, deduplicate on XLSX_DEDUP_COLS, and push back. """ logger.info("[Trimika XLSX] Reading new file: %s", new_file_path) df_new = pd.read_excel(new_file_path, dtype=str) df_new = df_new.fillna("") logger.info("[Trimika XLSX] New file rows: %d", len(df_new)) # Download existing file from GitHub existing_bytes = download_from_github(XLSX_FILE) if existing_bytes: df_existing = pd.read_excel(io.BytesIO(existing_bytes), dtype=str).fillna("") logger.info("[Trimika XLSX] Existing GitHub file rows: %d", len(df_existing)) else: df_existing = pd.DataFrame() logger.info("[Trimika XLSX] No existing file — starting fresh") # Append and deduplicate df_combined = pd.concat([df_existing, df_new], ignore_index=True) before = len(df_combined) # Only dedup on columns that actually exist in the data dedup_cols = [c for c in XLSX_DEDUP_COLS if c in df_combined.columns] if dedup_cols: df_combined = df_combined.drop_duplicates(subset=dedup_cols, keep="last") dupes = before - len(df_combined) if dupes: logger.info("[Trimika XLSX] Removed %d duplicate rows (kept newest)", dupes) else: logger.warning("[Trimika XLSX] Dedup columns not found: %s", XLSX_DEDUP_COLS) df_combined = df_combined.reset_index(drop=True) logger.info("[Trimika XLSX] Combined rows after dedup: %d", len(df_combined)) # Write to bytes and push buf = io.BytesIO() with pd.ExcelWriter(buf, engine="openpyxl") as writer: df_combined.to_excel(writer, index=False, sheet_name="wy_trimika_tool") buf.seek(0) return push_to_github( XLSX_FILE, buf.read(), f"Append trimika data — {len(df_new)} new rows, {len(df_combined)} total", ) # ══════════════════════════════════════════════════════════ # EDI CSV: APPEND + DEDUP + PUSH # ══════════════════════════════════════════════════════════ # All columns the HTML tool needs from the EDI CSV EDI_KEEP_COLS = [ "BROKERREFERENCE", "ENTRYNUMBER", "IMPORTERNAME", "IMPORTERID", "LINENUMBER", "HTS_NUMBER", "HTS_DESCRIPTION", "MANUFACTURERNAME", "ENTRYDATE", "ARRIVALDATE", "DATEOFEXPORT", "MODEOFTRANSPORT", "ENTEREDVALUE", "COUNTRYOFORIGIN", "ADL1_HTS_NUMBER", "ADL2_HTS_NUMBER", "ADL3_HTS_NUMBER", "ADL4_HTS_NUMBER", "ADL5_HTS_NUMBER", "ADL6_HTS_NUMBER", "ADL7_HTS_NUMBER", "ADL1_HTS_DESCRIPTION", "ADL2_HTS_DESCRIPTION", "ADL3_HTS_DESCRIPTION", "ADL4_HTS_DESCRIPTION", "ADL5_HTS_DESCRIPTION", "ADL6_HTS_DESCRIPTION", "ADL7_HTS_DESCRIPTION", ] def _slim_edi_df(df: pd.DataFrame) -> pd.DataFrame: """ Keep only the 28 columns needed by the HTML tool. Case-insensitive column matching. Missing columns are added as empty. """ col_map = {c.upper(): c for c in df.columns} result = pd.DataFrame() for needed in EDI_KEEP_COLS: actual = col_map.get(needed.upper()) if actual: result[needed] = df[actual].astype(str).str.replace(r"^'", "", regex=True).str.strip() else: logger.warning("[Trimika EDI] Column '%s' not found in file — leaving blank", needed) result[needed] = "" return result def append_and_push_edi(new_file_path: str) -> bool: """ Download edi_current.csv from GitHub, append today's rows (28 columns), deduplicate on EDI_DEDUP_COLS, and push back. """ logger.info("[Trimika EDI] Reading new CSV: %s", new_file_path) try: df_raw = pd.read_csv(new_file_path, dtype=str, low_memory=False).fillna("") except Exception as e: logger.error("[Trimika EDI] Failed to read CSV: %s", e) return False logger.info("[Trimika EDI] Raw file: %d rows, %d columns", len(df_raw), len(df_raw.columns)) # Slim to 28 needed columns df_new = _slim_edi_df(df_raw) logger.info("[Trimika EDI] Slimmed to %d columns, %d rows", len(df_new.columns), len(df_new)) # Download existing file from GitHub existing_bytes = download_from_github(EDI_FILE) if existing_bytes: try: df_existing = pd.read_csv(io.BytesIO(existing_bytes), dtype=str).fillna("") logger.info("[Trimika EDI] Existing GitHub file: %d rows", len(df_existing)) except Exception as e: logger.warning("[Trimika EDI] Could not read existing file (%s) — starting fresh", e) df_existing = pd.DataFrame(columns=EDI_KEEP_COLS) else: df_existing = pd.DataFrame(columns=EDI_KEEP_COLS) logger.info("[Trimika EDI] No existing file — starting fresh") # Append and deduplicate df_combined = pd.concat([df_existing, df_new], ignore_index=True) before = len(df_combined) dedup_cols = [c for c in EDI_DEDUP_COLS if c in df_combined.columns] if dedup_cols: df_combined = df_combined.drop_duplicates(subset=dedup_cols, keep="last") dupes = before - len(df_combined) if dupes: logger.info("[Trimika EDI] Removed %d duplicate rows (kept newest)", dupes) else: logger.warning("[Trimika EDI] Dedup columns not found: %s", EDI_DEDUP_COLS) df_combined = df_combined.reset_index(drop=True) logger.info("[Trimika EDI] Combined rows after dedup: %d", len(df_combined)) buf = io.BytesIO() df_combined.to_csv(buf, index=False) buf.seek(0) return push_to_github( EDI_FILE, buf.read(), f"Append EDI data — {len(df_new)} new rows, {len(df_combined)} total", ) # ══════════════════════════════════════════════════════════ # CLEANUP # ══════════════════════════════════════════════════════════ def cleanup(local_path: str): try: if local_path and os.path.exists(local_path): os.remove(local_path) logger.info("[Trimika] Removed temp file %s", local_path) except Exception as exc: logger.warning("[Trimika] Could not remove temp file: %s", exc) # ══════════════════════════════════════════════════════════ # MAIN ENTRY POINTS # ══════════════════════════════════════════════════════════ def run_trimika_update(): """ Full daily pipeline — XLSX then EDI. Called by APScheduler at 07:32 weekdays. """ logger.info("[Trimika] ─── Starting daily update ───") # ── XLSX ────────────────────────────────────────────── xlsx_path = get_attachment_from_outlook(EMAIL_SUBJECT, ".xlsx") if not xlsx_path: logger.error("[Trimika] Aborting XLSX update — no attachment found") _send_error_email( f"Could not retrieve XLSX from Outlook.\n" f"Check that an email with subject '{EMAIL_SUBJECT}' " f"exists in {SHARED_MAILBOX} / {REPORTS_FOLDER}." ) else: success = append_and_push_xlsx(xlsx_path) cleanup(xlsx_path) if success: logger.info("[Trimika] ─── XLSX update complete ✓ ───") else: logger.error("[Trimika] ─── XLSX update FAILED ───") _send_error_email("GitHub push failed for trimika_data.xlsx.") # ── EDI CSV ─────────────────────────────────────────── logger.info("[Trimika] ─── Starting EDI update ───") edi_path = get_attachment_from_outlook(EDI_SUBJECT, ".csv") if not edi_path: logger.error("[Trimika] Aborting EDI update — no attachment found") else: success = append_and_push_edi(edi_path) cleanup(edi_path) if success: logger.info("[Trimika] ─── EDI update complete ✓ ───") else: logger.error("[Trimika] ─── EDI update FAILED ───") # ══════════════════════════════════════════════════════════ # ERROR EMAIL # ══════════════════════════════════════════════════════════ def _send_error_email(body: str): pythoncom.CoInitialize() try: outlook = win32com.client.Dispatch("Outlook.Application") mail = outlook.CreateItem(0) mail.To = ERROR_NOTIFY mail.Subject = "[IFF Automation] Trimika Update Error" mail.Body = f"Trimika daily update encountered an error:\n\n{body}\n\nCheck logs for details." mail.Send() logger.info("[Trimika] Error notification sent to %s", ERROR_NOTIFY) except Exception as exc: logger.warning("[Trimika] Could not send error email: %s", exc) finally: pythoncom.CoUninitialize() #
